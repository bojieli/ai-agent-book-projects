WITH latest_salaries AS (
    SELECT DISTINCT ON (emp_id) emp_id, salary
    FROM salaries
    ORDER BY emp_id, pay_date DESC
)

    CASE 
        WHEN (CURRENT_DATE - e.hire_date)::INTEGER < 365 THEN '入职一年内'

        WHEN (CURRENT_DATE - e.hire_date)::INTEGER BETWEEN 730 AND 1094 THEN '两到三年'

    AVG(ls.salary) AS average_latest_salary
FROM employees e
